enum 與 match

定義與建立

enum LineKind {
    Empty,
    Comment,
    Code(String),   // 變體可以帶資料，也可以不帶
}

let a = LineKind::Empty;          // 建立值用 ::
let b = LineKind::Code("fn main".to_string());

struct 是「且」，enum 是「或」

struct Report { filename: String, content: String }  // 同時擁有全部欄位
enum LineKind { Empty, Comment, Code(String) }        // 只會是其中一個變體

match：一定要列完所有變體

match kind {
    LineKind::Empty => println!("空行"),
    LineKind::Comment => println!("註解"),
    LineKind::Code(text) => println!("程式碼：{}", text),  // 順便取出資料
}
// 少列一個變體 => 編譯失敗，不是執行期才出錯

你早就在用的 enum

Option<T>   =   Some(T)  |  None
Result<T, E> =  Ok(T)    |  Err(E)

match opt {
    Some(value) => ...,   // 帶資料的變體
    None => ...,          // 空變體
}
